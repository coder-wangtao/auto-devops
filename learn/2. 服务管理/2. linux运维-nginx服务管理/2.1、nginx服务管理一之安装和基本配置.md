2、nginx的基本配置

\#vim  /usr/local/nginx/conf/nginx.conf

![image-20240629070004996](/Users/hellen/Library/Application Support/typora-user-images/image-20240629070004996.png)

user nginx  // nginx进程归nginx用户

worker_processes  1   

//nginx要开启的进程数，有几个CPU就写几个，值等于cat /proc/cpuinfo | grep processor |wc  -l

worker_rlimit_nofile  65535  //nginx worker 进程能打开的最大文件数

worker_connections 10240  //最大并发数值

pid  /var/run/nginx.pid   //这一行主要是为了后面配合nginx的管理脚本

注意：行后面的分号不要忘了



配置调整好后，进行语法检查，无误后，启动服务

\#/usr/loca/nginx/sbin/nginx  -t   //语法检查

\#/usr/local/nginx/sbin/nginx    //启动nginx服务; -s stop/ -s reopen /-s reload

-s reopen  重新打开日志文件

-s reload  重新加载配置文件

-s stop   关闭nginx服务



\#netstat  -ntpl  |grep  :80   //有看到nginx进程和端口，就可以到浏览器访问看看



3、部署nginx管理脚本

\#将脚本nginx上传到/etc/init.d目录上

\#vim  /etc/init.d/nginx  //内容如下,文件也放在了目录下，可以直接拷贝使用。

\#!/bin/bash

\# nginx Startup script for the Nginx HTTP Server

\# it is v.0.0.2 version.

\# chkconfig: - 85 15

\# description: Nginx is a high-performance web and proxy server.

\#        It has a lot of features, but it's not for everyone.

\# processname: nginx

\# pidfile: /var/run/nginx.pid

\# config: /usr/local/nginx/conf/nginx.conf

nginxd=/usr/local/nginx/sbin/nginx

nginx_config=/usr/local/nginx/conf/nginx.conf

nginx_pid=/var/run/nginx.pid

RETVAL=0

prog="nginx"

\# Source function library.

. /etc/rc.d/init.d/functions

\# Source networking configuration.

. /etc/sysconfig/network

\# Check that networking is up.

\#[ ${NETWORKING} = "no" ] && exit 0

[ -x $nginxd ] || exit 0

\# Start nginx daemons functions.

start() {

if [ -e $nginx_pid ];then

  echo "nginx already running...."

  exit 1

fi

  echo -n $"Starting $prog: "

  daemon $nginxd -c ${nginx_config}

  RETVAL=$?

  echo

  [ $RETVAL = 0 ] && touch /var/lock/subsys/nginx

  return $RETVAL

}
